@st.cache_resource
def carregar_recursos_globais():
    meta = pd.read_csv(os.path.join(data_dir, "node_metadata.csv"))
    adj_mx = np.load(os.path.join(data_dir, "adj_mx.npy"))
    adj_tensor = torch.tensor(adj_mx, dtype=torch.float32).to(device)
    
    model = STGCN_TrafficModel(
        num_nodes=len(meta), num_features=2, num_timesteps_input=12, num_timesteps_output=12, adj_mx=adj_tensor
    ).to(device)
    
    caminho_pesos = os.path.join(modelo_dir, "best_stgcn_chicago.pt")
    if os.path.exists(caminho_pesos):
        model.load_state_dict(torch.load(caminway_weights := torch.load(caminho_pesos, map_location=device)))
    else:
        st.warning("Aviso: Pesos do modelo não localizados. Operando com inicialização aleatória.")
    model.eval()
    return model, meta

def geocodificar_endereco(endereco):
    """ Busca coordenadas lat/lon protegendo a requisição e sanitizando o input. """
    url = "https://nominatim.openstreetmap.org/search"
    
    # Substituição de caracteres comerciais problemáticos para o Nominatim
    # Apenas fazemos a atribuição simples aqui
    endereco_sanitizado = endereco.replace("&", "and")
    
    params = {'q': endereco_sanitizado + ", Chicago, IL", 'format': 'json', 'limit': 1}
    headers = {'User-Agent': 'RouteStation_IC_Academic_Project/3.0 (academic_test@domain.com)'}
    
    try:
        response = requests.get(url, params=params, headers=headers)
        if response.status_code == 200 and len(response.json()) > 0:
            res = response.json()[0]
            return float(res['lat']), float(res['lon'])
    except Exception as e:
        st.error(f"Falha de comunicação com o servidor GIS Nominatim: {e}")
    return None, None

def encontrar_no_mais_proximo(lat_usuario, lon_usuario, meta_df):
    lat1 = np.radians(lat_usuario)
    lon1 = np.radians(lon_usuario)
    lat2 = np.radians(meta_df['START_LATITUDE'].values)
    lon2 = np.radians(meta_df['START_LONGITUDE'].values)
    dlon = lon2 - lon1
    dlat = lat2 - lat1
    a = np.sin(dlat/2)**2 + np.cos(lat1) * np.cos(lat2) * np.sin(dlon/2)**2
    return int(np.argmin(6371.0 * (2 * np.arcsin(np.sqrt(a)))))
